<objective>
Close the reassignment loop: when a task moves to a different teammate, the old personalized description is invalidated immediately (so the new assignee never sees text written for someone else, even for one cron cycle), and a fresh `task_reframe` job is enqueued. Lock the FRAME-06 (tone bounds) and FRAME-07 (no external inference) contracts with golden tests that defend against future prompt drift.

Purpose: Plans 01 and 02 establish forward generation + read. Without invalidation, a reassigned task would show the new assignee a description tailored to the prior person — a leak in the wrong direction (the new assignee gets context about the old one, and the framing tone is wrong). Plus the golden tests pin the boundary so future prompt edits can't silently weaken the contract.

Output: Updated `reassignTask` with invalidation + enqueue, three new test files (1 invalidation regression + 2 golden), no schema changes.
</objective>

    ```typescript
    export async function enqueueReframeJob(
      taskId: string,
      assigneeTeammateId: string,
      teamId: string,
    ): Promise<void> {
      // teammate → userId resolution
      const teammate = await getTeammate(assigneeTeammateId);
      if (!teammate?.userId) {
        logger.debug('task_reframe enqueue skipped — teammate has no userId', { assigneeTeammateId });
        return;
      }
      try {
        await enqueueJob({
          teamId,
          userId: teammate.userId,
          kind: 'task_reframe',
          payload: { taskId, assigneeUserId: teammate.userId, teamId },
        });
      } catch (err) {
        logger.warn('task_reframe enqueue failed', {
          taskId,
          assigneeUserId: teammate.userId,
          err: err instanceof Error ? err.message : String(err),
        });
      }
    }
    ```

    ```typescript
    export async function reassignTask(
      taskId: string,
      teammateId: string | null,
      assignedBy: 'recgon' | string,
      schedule?: { ... },
    ): Promise<void> {
      // Read current state to detect actual reassignment vs no-op
      const { data: current } = await supabase
        .from('agent_tasks')
        .select('assigned_to, team_id')
        .eq('id', taskId)
        .maybeSingle();
      const previousAssignedTo = (current?.assigned_to as string | null) ?? null;
      const teamId = (current?.team_id as string) ?? '';
      const isActualReassignment =
        previousAssignedTo !== teammateId && (previousAssignedTo !== null || teammateId !== null);

      const update: Record<string, unknown> = {
        assigned_to: teammateId,
        assigned_by: assignedBy,
        assigned_at: teammateId ? new Date().toISOString() : null,
        status: teammateId ? 'assigned' : 'unassigned',
        job_id: null,
      };
      if (!teammateId) {
        update.scheduled_date = null;
        update.scheduled_until_date = null;
        update.schedule_note = null;
      }
      if (schedule?.scheduledDate !== undefined) update.scheduled_date = schedule.scheduledDate;
      if (schedule?.scheduledUntilDate !== undefined) update.scheduled_until_date = schedule.scheduledUntilDate;
      if (schedule?.deadline !== undefined) update.deadline = schedule.deadline;
      if (schedule?.scheduleNote !== undefined) update.schedule_note = schedule.scheduleNote;

      // Phase 4 / Plan 03 — invalidate personalized columns on actual reassignment.
      // FRAME-04: the new assignee must NEVER see text written for the old one,
      // not even for a single cron cycle. Atomic: same update statement.
      if (isActualReassignment) {
        update.personalized_description = null;
        update.personalized_description_for_user_id = null;
      }

      const { error } = await supabase.from('agent_tasks').update(update).eq('id', taskId);
      if (error) throw new Error(`reassignTask failed: ${error.message}`);

      // Post-update: enqueue a new reframe job when there's a new assignee to personalize for.
      // Fire-and-forget. enqueueReframeJob catches its own failures.
      if (isActualReassignment && teammateId !== null) {
        await enqueueReframeJob(taskId, teammateId, teamId);
      }
    }
    ```
