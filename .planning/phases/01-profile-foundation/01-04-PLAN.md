    function applyProfileMerge<T extends { userId: string | null; fitProfile: any }>(
      teammates: T[],
      profiles: TeammateProfile[],
    ): Array<T & { interests?: string[] }> {
      const byUserId = new Map(profiles.map(p => [p.userId, p]));
